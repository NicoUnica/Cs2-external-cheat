cmake_minimum_required(VERSION 3.20)
project(NicoNucleo LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Este núcleo requiere Windows x64.")
endif()
add_library(nico STATIC nucleo/juego.cpp nucleo/datos.cpp)
target_compile_features(nico PUBLIC cxx_std_20)
target_include_directories(nico PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/nucleo")
target_compile_definitions(nico PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(nico PUBLIC /utf-8 PRIVATE /W3)
endif()
target_link_libraries(nico PUBLIC user32 psapi)
